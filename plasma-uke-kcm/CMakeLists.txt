cmake_minimum_required(VERSION 3.22)
project(plasma-uke-kcm VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(KDECMakeSettings)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Qml Quick Test)
find_package(KF6KCMUtils 6.0 REQUIRED)
find_package(KF6CoreAddons 6.0 REQUIRED)
find_package(KF6I18n 6.0 REQUIRED)
find_package(Gettext REQUIRED)
gettext_process_po_files(tr ALL PO_FILES po/tr.po)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/tr.gmo DESTINATION ${KDE_INSTALL_LOCALEDIR}/tr/LC_MESSAGES RENAME kcm_uke.mo)
kcmutils_add_qml_kcm(kcm_uke SOURCES src/kcm.cpp src/overview-model.cpp)
target_link_libraries(kcm_uke PRIVATE KF6::KCMUtilsQuick Qt6::Quick)
target_compile_options(kcm_uke PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(overview-model-test tests/model.cpp src/overview-model.cpp)
    target_include_directories(overview-model-test PRIVATE src)
    target_link_libraries(overview-model-test PRIVATE Qt6::Core Qt6::Test)
    target_compile_options(overview-model-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME overview-model COMMAND overview-model-test)
    add_executable(render-kcm tests/render.cpp)
    target_link_libraries(render-kcm PRIVATE KF6::KCMUtilsQuick KF6::I18n Qt6::Quick)
    target_compile_options(render-kcm PRIVATE -Wall -Wextra -Werror)
endif()
