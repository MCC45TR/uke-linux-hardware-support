cmake_minimum_required(VERSION 3.20)
project(uke-control-gtk VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.12 libadwaita-1>=1.5)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET jsoncpp)
set(UKE_CONTROL_SOURCE "" CACHE PATH "Optional sibling uke-control source tree for development")
if(UKE_CONTROL_SOURCE)
  find_library(UKE_CLIENT NAMES libuke-control-client.so.1 PATHS "${UKE_CONTROL_SOURCE}/build" NO_DEFAULT_PATH REQUIRED)
  add_library(uke-client SHARED IMPORTED)
  target_compile_definitions(uke-client INTERFACE UKE_CONTROL_DEVELOPMENT_CLIENT=1)
  set_target_properties(uke-client PROPERTIES IMPORTED_LOCATION "${UKE_CLIENT}" INTERFACE_INCLUDE_DIRECTORIES "${UKE_CONTROL_SOURCE}/src")
else()
  pkg_check_modules(CLIENT REQUIRED IMPORTED_TARGET uke-control-client)
  add_library(uke-client INTERFACE)
  target_link_libraries(uke-client INTERFACE PkgConfig::CLIENT)
endif()
add_executable(uke-control-gtk src/main.cpp)
target_compile_options(uke-control-gtk PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(uke-control-gtk PRIVATE uke-client PkgConfig::GTK PkgConfig::JSON Threads::Threads)
install(TARGETS uke-control-gtk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/org.uke.Control.Gtk.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/org.uke.Control.Gtk.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
 add_executable(gtk-fixture-test tests/fixture.cpp)
 target_link_libraries(gtk-fixture-test PRIVATE uke-client PkgConfig::JSON)
 target_include_directories(gtk-fixture-test PRIVATE src)
 add_test(NAME fixture-parser COMMAND gtk-fixture-test)
endif()
